# names, and write-expected.txt holds what each leaves. The operation language is in
# write_compare.py's docstring. Paths under /files, /extents, /htree, /links, /attributes and
# /empty-directory are the corpus's own, written by mke2fs, so a case that touches one changes a
# file Linux wrote.
#
# See nifeos/record:design/roadmap/0190-ext4-read-write/write-oracle.md.
#

# Found by this oracle. At 1 KiB blocks a group holds 1,024 inodes and its inode bitmap block 8,192
# bits; the fourth new inode is the first in group 3, which mke2fs left INODE_UNINIT, and rsext4
# writes that group's bitmap without setting the padding bits past the group's last inode.
case inode-bitmap-padding
fill / 4 group-

# A directory rsext4 grows from one block into an htree. With empty files its blocks stay
# contiguous and its extents stay in the inode.
case htree-growth
mkdir /grow
fill /grow 3000 entry-
fill /htree 500 added-

# Bug 1. File data interleaved with the directory's growth fragments its blocks, so it passes the
# inode's four extents into a level-1 leaf, and its i_blocks comes out short.
case directory-past-inline-extents
mkdir /grow
fill /grow 1000 entry- 4096 18

# Bug 2. A file of 1,400 separate extents has a depth-2 tree at 1 KiB and 4 KiB blocks; truncating
# it to 0 fails.
case truncate-depth-2-to-zero
create /cut
stripes /cut 1400 8192 4096 19
truncate /cut 0

# Bug 2, its atomicity half: the failed truncation must leave the file's bytes as they were.
case truncate-depth-2-inside-a-block
create /cut
stripes /cut 1400 8192 4096 20
truncate /cut 100

case truncate-linux-depth-2
truncate /extents/depth-2 0
